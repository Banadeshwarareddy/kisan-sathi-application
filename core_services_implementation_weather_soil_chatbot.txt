import requests
from django.conf import settings
from django.core.cache import cache

def get_weather_forecast(city):
    # Check cache first
    cache_key = f"weather_{city.lower().replace(' ', '_')}"
    cached_data = cache.get(cache_key)
    if cached_data:
        return cached_data

    api_key = settings.OPENWEATHERMAP_API_KEY
    # OpenWeatherMap API call (Simplified for demonstration)
    url = f"https://api.openweathermap.org/data/2.5/forecast?q={city}&units=metric&appid={api_key}"
    
    try:
        # response = requests.get(url)
        # data = response.json()
        
        # Mock data for production-ready logic structure
        data = {
            'city': city,
            'current': {'temp': 28, 'condition': 'Sunny', 'humidity': 45, 'wind': 12},
            'forecast': [
                {'day': 'Mon', 'temp': 27, 'condition': 'Partly Cloudy'},
                {'day': 'Tue', 'temp': 29, 'condition': 'Sunny'},
                {'day': 'Wed', 'temp': 26, 'condition': 'Rainy'},
            ]
        }
        # Cache for 1 hour
        cache.set(cache_key, data, 3600)
        return data
    except Exception as e:
        print(f"Weather API Error: {e}")
        return None

        ]
    }

        recommendations['status'] = 'Acidic'
        recommendations['fertilizers'].append({'name': 'Lime', 'quantity': '200kg/acre', 'timing': 'Soil preparation'})
    
    return recommendations
